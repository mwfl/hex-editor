cmake_minimum_required(VERSION 3.21)
project(hex_editor VERSION 0.1.2 LANGUAGES CXX)

include(FetchContent)
option(HEX_EDITOR_USE_LOCAL_MWFL "Use a local mwfl checkout" OFF)
set(HEX_EDITOR_MWFL_SOURCE_DIR "../mwfl" CACHE PATH "Local mwfl source directory")
set(MWFL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MWFL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(HEX_EDITOR_USE_LOCAL_MWFL)
  add_subdirectory("${HEX_EDITOR_MWFL_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/mwfl")
else()
  FetchContent_Declare(mwfl GIT_REPOSITORY https://github.com/mwfl/mwfl.git GIT_TAG v0.1.3)
  FetchContent_MakeAvailable(mwfl)
endif()

add_executable(hex-editor WIN32 main.cpp hex_document.cpp hex_document.h hex_view.cpp hex_view.h hex_editor.rc resource.h app-icon.ico app.manifest)
target_link_libraries(hex-editor PRIVATE mwfl::ui shell32)
target_compile_features(hex-editor PRIVATE cxx_std_20)
target_compile_definitions(hex-editor PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)

enable_testing()
add_executable(hex-editor-tests hex_document_test.cpp hex_document.cpp hex_document.h)
target_compile_features(hex-editor-tests PRIVATE cxx_std_20)
target_compile_definitions(hex-editor-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
foreach(target hex-editor hex-editor-tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  endif()
endforeach()
add_test(NAME hex-editor.model COMMAND hex-editor-tests)
add_test(NAME hex-editor.gui COMMAND hex-editor --self-test)
set_tests_properties(hex-editor.gui PROPERTIES TIMEOUT 20)

install(TARGETS hex-editor RUNTIME DESTINATION . COMPONENT Application)
install(FILES README.md LICENSE DESTINATION . COMPONENT Application)
